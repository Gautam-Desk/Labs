Q1. Personal Data Form

<!DOCTYPE html>
<html>
<head>
  <title>Personal Data Form</title>
  <style>
    body { font-family: Arial; background: #f3f3f3; }
    form { width: 420px; margin: 40px auto; background: white; padding: 20px; border-radius: 10px; }
    input, select, textarea { width: 100%; padding: 8px; margin: 6px 0 12px; }
    .row { margin-bottom: 10px; }
  </style>
</head>
<body>
  <form>
    <h2>Personal Data Form</h2>
    <label>Name:</label>
    <input type="text" name="name" >

    <label>Password:</label>
    <input type="password" name="pass" >

    <label>Email:</label>
    <input type="email" name="email" >

    <label>City:</label>
    <select>
      <option>Mumbai</option>
      <option>Delhi</option>
      <option>Pune</option>
    </select>

    <label>Gender:</label>
    <input type="radio" name="gender" value="Male"> Male
    <input type="radio" name="gender" value="Female"> Female

    <label>Hobbies:</label>
    <input type="checkbox" name="hobby" value="Reading"> Reading
    <input type="checkbox" name="hobby" value="Music"> Music
    <input type="checkbox" name="hobby" value="Sports"> Sports

    <input type="submit" value="Submit">
  </form>
</body>
</html>

Q7. My City Page

<html>
<head>
  <title>myCity</title>
  <style>
    body { text-align: center; font-family: Arial; }
    h1 { color: blue; font-size: 50px; }
    .landmark { font-weight: bold; margin: 10px; }
    .red { color: red; }
    .green { color: green; }
    .purple { color: purple; }

Q13. Warning Message

<!DOCTYPE html>
<html>
<head>
  <style>
    body { background: red; text-align: center; color: white; }
    .warning {
      font-size: 60px; font-weight: bold;
      animation: blink 1s infinite;
    }
    @keyframes blink { 0%, 100% { opacity: 1; } 50% { opacity: 0; } }
  </style>
</head>
<body>
  <div class="warning">WARNING</div>
  <marquee>Read the message</marquee>
</body>
</html>
